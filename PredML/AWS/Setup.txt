pipeline {
    agent any

    triggers {
        // Run every Sunday at midnight
        cron('0 0 * * 0')
        // Also triggers on git push automatically via webhook
    }

    environment {
        AWS_REGION     = 'ap-southeast-2'
        AWS_ACCOUNT_ID = '<your-account-id>'
    }

    stages {
        stage('Build & Push predml-batch') {
            // Only rebuild image if code changed
            when { changeset "src/**" }
            steps {
                sh '''
                    aws ecr get-login-password --region $AWS_REGION | \
                    docker login --username AWS --password-stdin \
                    $AWS_ACCOUNT_ID.dkr.ecr.$AWS_REGION.amazonaws.com

                    docker build -t predml-batch:latest -f docker/Dockerfile.batch .
                    docker push \
                    $AWS_ACCOUNT_ID.dkr.ecr.$AWS_REGION.amazonaws.com/predml-batch:latest
                '''
            }
        }

        stage('Trigger Training Pipeline') {
            steps {
                sh '''
                    aws sagemaker start-pipeline-execution \
                        --pipeline-name PredML-Training-Pipeline \
                        --region $AWS_REGION
                '''
            }
        }
    }
}

from sagemaker.workflow.pipeline import Pipeline
from sagemaker.workflow.steps import ProcessingStep
from sagemaker.processing import ScriptProcessor, ProcessingInput, ProcessingOutput

def make_processor(stage: str):
    return ScriptProcessor(
        image_uri=ECR_BATCH_IMAGE,
        role=ROLE,
        instance_type="ml.m5.xlarge",
        instance_count=1,
        env={
            "STAGE":               stage,
            "S3_BUCKET":           S3_BUCKET,
            "MLFLOW_TRACKING_URI": MLFLOW_URI,
            "AWS_DEFAULT_REGION":  REGION
        }
    )

# Define steps
step_preprocess    = ProcessingStep(name="Preprocess",
                        processor=make_processor("01_preprocess"), ...)
step_feature_eng   = ProcessingStep(name="FeatureEngineering",
                        processor=make_processor("02_feature_engineering"), ...)
step_feature_store = ProcessingStep(name="FeatureStore",
                        processor=make_processor("03_feature_store"), ...)
step_train         = ProcessingStep(name="TrainModel",
                        processor=make_processor("04_train_model"), ...)
step_register      = ProcessingStep(name="RegisterModel",
                        processor=make_processor("05_register_model"), ...)

# Training pipeline — stages 01 to 05 only
training_pipeline = Pipeline(
    name="PredML-Training-Pipeline",
    steps=[
        step_preprocess,
        step_feature_eng,
        step_feature_store,
        step_train,
        step_register

)
training_pipeline.upsert(role_arn=ROLE)

3. EventBridge Schedules for Batch & Monitoring
Use Amazon EventBridge to trigger SageMaker pipelines on a schedule — no Jenkins needed for these.

Step-by-step in Console:
Open EventBridge Console  → Schedules → Create schedule

For Batch Inference (runs daily at 2 AM):

Schedule name: predml-batch-inference-schedule
Schedule pattern: Recurring schedule → Cron: 0 2 * * ? *
Target: SageMaker Pipeline → select PredML-Batch-Inference-Pipeline
IAM role: create new (EventBridge needs permission to start pipeline)
For Monitoring (runs daily at 4 AM — after batch completes):

Schedule name: predml-monitoring-schedule
Schedule pattern: Cron: 0 4 * * ? *
Target: SageMaker Pipeline → select PredML-Monitoring-Pipeline
4. Auto-trigger Rollback from Monitoring
If 08_model_monitoring.py detects drift, it can automatically trigger the rollback job:

# Inside 08_model_monitoring.py — trigger rollback if drift detected
import boto3

def trigger_rollback_if_needed(drift_score: float, threshold: float = 0.15):
    if drift_score > threshold:
        print(f"Drift detected ({drift_score:.3f}) — triggering rollback pipeline")
        sm_client = boto3.client("sagemaker", region_name="ap-southeast-2")
        sm_client.start_pipeline_execution(
            PipelineName="PredML-Rollback-Pipeline"
        )

Jenkins Pipeline for Terraform
jenkins/Jenkinsfile.terraform
pipeline {
    agent any

    environment {
        AWS_ACCESS_KEY_ID     = credentials('AWS_ACCESS_KEY_ID')
        AWS_SECRET_ACCESS_KEY = credentials('AWS_SECRET_ACCESS_KEY')
        AWS_REGION            = 'ap-southeast-2'
        TF_VAR_s3_bucket_name = 'predml-data-ap-southeast-2'
    }

    parameters {
        choice(
            name: 'TF_ACTION',
            choices: ['plan', 'apply', 'destroy'],
            description: 'Terraform action to run'
        )
    }

    stages {
        stage('Checkout') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/<your-org>/predml-infra.git',
                    credentialsId: 'github-credentials'
            }
        }

        stage('Install Terraform') {
            steps {
                sh '''
                    if ! command -v terraform &> /dev/null; then
                        curl -fsSL https://releases.hashicorp.com/terraform/1.7.0/terraform_1.7.0_linux_amd64.zip \
                            -o terraform.zip
                        unzip terraform.zip
                        sudo mv terraform /usr/local/bin/
                    fi
                    terraform version
                '''
            }
        }

        stage('Terraform Init') {
            steps {
                sh '''
                    terraform init \
                        -backend-config="bucket=predml-terraform-state-ap-southeast-2" \
                        -backend-config="key=predml/terraform.tfstate" \
                        -backend-config="region=$AWS_REGION"
                '''
            }
        }

        stage('Terraform Validate') {
            steps {
                sh 'terraform validate'
            }
        }

        stage('Terraform Plan') {
            steps {
                sh '''
                    terraform plan \
                        -var-file="terraform.tfvars" \
                        -out=tfplan
                '''
            }
        }

        stage('Approval — Apply/Destroy Only') {
            when {
                expression { params.TF_ACTION in ['apply', 'destroy'] }
            }
            steps {
                // Pauses pipeline — requires manual approval in Jenkins UI
                input message: "Approve Terraform ${params.TF_ACTION}?",
                      ok: "Yes, proceed"
            }
        }

        stage('Terraform Apply') {
            when {
                expression { params.TF_ACTION == 'apply' }
            }
            steps {
                sh 'terraform apply -auto-approve tfplan'
            }
        }

        stage('Terraform Destroy') {
            when {
                expression { params.TF_ACTION == 'destroy' }
            }
            steps {
                sh '''
                    terraform destroy \
                        -var-file="terraform.tfvars" \
                        -auto-approve
                '''
            }
        }
    }

    post {
        success {
            echo "Terraform ${params.TF_ACTION} completed successfully"
        }
        failure {
            echo "Terraform ${params.TF_ACTION} failed — check logs"
        }
    }
}
